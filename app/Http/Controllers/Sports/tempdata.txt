private function extractPlayerData(array $squadData, int $teamId, int $matchId): array
    {
        return array_filter(array_map(function ($data) use ($teamId, $matchId) {
            // Add additional validation to ensure all required fields exist
            if (!$this->validatePlayerData($data)) {
                Log::warning("Invalid player data", [
                    'team_id' => $teamId,
                    'match_id' => $matchId,
                    'player_data' => $data
                ]);
                return null;
            }

            return [
                'player_id' => $data['id'],
                'team_id' => $teamId,
                'match_id' => $matchId,
                'country_id' => $data['country_id'] ?? null,
                'firstname' => $data['firstname'] ?? '',
                'lastname' => $data['lastname'] ?? '',
                'fullname' => $data['fullname'] ?? '',
                'image_path' => $data['image_path'] ?? '',
                'dateofbirth' => $data['dateofbirth'] ?? null,
                'gender' => $data['gender'] ?? '',
                'battingstyle' => $data['battingstyle'] ?? '',
                'bowlingstyle' => $data['bowlingstyle'] ?? '',
                'position_id' => $data['position']['id'] ?? null,
                'position_name' => $data['position']['name'] ?? '',
                'updated_at' => now(),
            ];
        }, $squadData));
    }

    private function validatePlayerData(array $data): bool
    {
        // Validate essential player data
        return isset($data['id']) &&
            isset($data['position']) &&
            is_array($data['position']);
    }

    private function bulkUpsertPlayers(array $playersData): int
    {
        if (empty($playersData)) {
            Log::warning('No players data to upsert');
            return 0;
        }

        try {
            $insertedCount = DB::transaction(function () use ($playersData) {
                return Player::upsert(
                    $playersData,
                    ['player_id', 'team_id', 'match_id'],
                    ['fullname', 'image_path', 'battingstyle', 'bowlingstyle', 'position_id', 'position_name', 'updated_at']
                );
            });

            Log::info('Players upserted successfully', [
                'total_players' => count($playersData),
                'inserted_count' => $insertedCount
            ]);

            return $insertedCount;
        } catch (\Exception $e) {
            Log::error("Bulk upsert failed: " . $e->getMessage(), [
                'players_data_count' => count($playersData)
            ]);
            throw $e;
        }
    }
